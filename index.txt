        /* 音效系統 (Audio Context) */
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        function playSound(type) {
            if (!soundEnabled) return;
            try {
                if (audioCtx.state === 'suspended') {
                    audioCtx.resume();
                }

                if (type === 'correct') {
                    // 浮誇大三和弦與星光琶音效果 (C5 - E5 - G5 - C6)
                    const now = audioCtx.currentTime;
                    const freqs = [523.25, 659.25, 783.99, 1046.50, 1318.51];
                    
                    freqs.forEach((freq, index) => {
                        const osc = audioCtx.createOscillator();
                        const gain = audioCtx.createGain();
                        osc.type = 'triangle';
                        osc.frequency.setValueAtTime(freq, now + index * 0.08);

                        gain.gain.setValueAtTime(0, now + index * 0.08);
                        gain.gain.linearRampToValueAtTime(0.25, now + index * 0.08 + 0.02);
                        gain.gain.exponentialRampToValueAtTime(0.001, now + index * 0.08 + 0.45);

                        osc.connect(gain);
                        gain.connect(audioCtx.destination);
                        osc.start(now + index * 0.08);
                        osc.stop(now + index * 0.08 + 0.45);
                    });
                } else if (type === 'wrong') {
                    const osc = audioCtx.createOscillator();
                    const gain = audioCtx.createGain();
                    osc.connect(gain);
                    gain.connect(audioCtx.destination);
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(220, audioCtx.currentTime);
                    osc.frequency.exponentialRampToValueAtTime(130.81, audioCtx.currentTime + 0.25);
                    gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.25);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.25);
                } else if (type === 'select') {
                    const osc = audioCtx.createOscillator();
                    const gain = audioCtx.createGain();
                    osc.connect(gain);
                    gain.connect(audioCtx.destination);
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(440, audioCtx.currentTime);
                    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);
                    osc.start();
                    osc.stop(audioCtx.currentTime + 0.1);
                }
            } catch (e) { console.log(e); }
        }

            soundEnabled = !soundEnabled;

            } else {
                icon.className = 'fa-solid fa-volume-xmark text-xs sm:text-sm text-red-400';
            }
        }

            selectedBopoText = null;
            selectedBopoElement = null;

            // 1. 進度與提示 UI
            document.getElementById('progress-text').innerText = `${currentIndex + 1} / 8`;

            document.getElementById('illustration-container').innerHTML = q.svg;

            const wordArea = document.getElementById('word-display-area');
            wordArea.innerHTML = '';

            const charArray = q.word.split('');
            charArray.forEach((char, idx) => {
                const isTarget = (idx === q.targetIndex);
                const charGroup = document.createElement('div');
                charGroup.className = 'flex items-center gap-1.5 sm:gap-3 bg-slate-50/90 px-2.5 sm:px-4 py-1.5 sm:py-2 rounded-2xl sm:rounded-3xl border border-slate-200/80 shadow-inner shrink-0';

                // 國字
                const hanziEl = document.createElement('span');
                hanziEl.className = 'text-4xl sm:text-6xl font-black text-slate-800 bopo-font tracking-tight';

                charGroup.appendChild(hanziEl);

                // 注音組合容器 (直式上下拼音)
                const pinyinStr = q.pinyin[idx];
                const toneMark = getToneMark(pinyinStr);

                    dropSlot.addEventListener('dragover', (e) => {
                        e.preventDefault();

                    });
                    dropSlot.addEventListener('dragleave', () => {
                        dropSlot.classList.remove('drag-over');
                    });
                    dropSlot.addEventListener('drop', (e) => {
                        e.preventDefault();
                        dropSlot.classList.remove('drag-over');
                        const droppedText = e.dataTransfer.getData('text/plain');
                        verifyAnswer(droppedText, dropSlot);
                    });

                    // 提示輕聲或聲調位置
                    if (toneMark) {
                        const toneEl = document.createElement('span');
                        toneEl.className = 'absolute -right-2 top-1 text-sm sm:text-lg text-amber-600 font-bold';
                        toneEl.innerText = toneMark;
                        dropSlot.appendChild(toneEl);
                    }

                    charGroup.appendChild(dropSlot);

                    // 一般已知拼音（傳統直式上下結構）
                    const cleanPinyin = removeTone(pinyinStr);
                    const parsed = parseVerticalBopo(cleanPinyin);

                    zhuyinContainer.className = 'zhuyin-vertical font-black bopo-font text-slate-700';

                        const yEl = document.createElement('span');
                        yEl.className = 'text-base sm:text-xl';
                        yEl.innerText = parsed.y;
                        symbolStack.appendChild(yEl);
                    }

                        const toneEl = document.createElement('span');
                        toneEl.className = 'text-xs sm:text-sm text-slate-500 font-bold ml-0.5 self-center';
                        toneEl.innerText = toneMark;
                        zhuyinContainer.appendChild(toneEl);
                    }

                    charGroup.appendChild(zhuyinContainer);
                }

                wordArea.appendChild(charGroup);
            });

            // 4. 產生兩側 2 個大字卡（左1 右1）
            generateOptionCards(q.answer);

            // 5. 語音朗讀
            setTimeout(() => speakCurrentWord(), 400);
        }

        /* 拆解聲母與韻母 (例: ㄅㄢ -> s:ㄅ, y:ㄢ) */
        function parseVerticalBopo(bopoStr) {
            if (BOPO_MAP[bopoStr]) {
                return BOPO_MAP[bopoStr];
            }
            if (bopoStr.startsWith('ㄅ')) {
                return { s: 'ㄅ', y: bopoStr.substring(1) };
            }
            return { s: bopoStr[0] || '', y: bopoStr[1] || '' };
        }

        /* 產生兩側大字卡 (左右各1個，共2選項) */
        function generateOptionCards(correctAnswer) {
            const leftContainer = document.getElementById('left-options-container');
            const rightContainer = document.getElementById('right-options-container');
            leftContainer.innerHTML = '';
            rightContainer.innerHTML = '';

            const distractor = ALL_BOPO.filter(b => b !== correctAnswer).sort(() => 0.5 - Math.random())[0];
            const twoOptions = [correctAnswer, distractor].sort(() => 0.5 - Math.random());

            leftContainer.appendChild(createTileElement(twoOptions[0], 'left-0'));
            rightContainer.appendChild(createTileElement(twoOptions[1], 'right-0'));
        }

        /* 建立巨型「直式上下拼音」選項卡 */
        function createTileElement(text, key) {
            const parsed = parseVerticalBopo(text);
            const tile = document.createElement('div');
            tile.className = 'w-full h-4/5 max-h-[280px] min-h-[160px] bg-gradient-to-b from-amber-50 to-amber-100/90 rounded-3xl border-4 border-amber-300 shadow-md hover:shadow-xl transition-all duration-200 cursor-grab active:cursor-grabbing flex flex-col items-center justify-center p-2 relative select-none';
            tile.setAttribute('draggable', 'true');
            tile.setAttribute('id', `tile-${key}`);

            // 直式上下拼音內容
            const stack = document.createElement('div');
            stack.className = 'option-bopo-stack bopo-font text-amber-900 font-black tracking-tighter';

            const sEl = document.createElement('span');
            sEl.innerText = parsed.s;
            const yEl = document.createElement('span');
            yEl.innerText = parsed.y;

            stack.appendChild(sEl);
            stack.appendChild(yEl);
            tile.appendChild(stack);

            // 拖曳事件
            tile.addEventListener('dragstart', (e) => {
                e.dataTransfer.setData('text/plain', text);
                tile.classList.add('opacity-50', 'scale-95');
                playSound('select');
            });

            tile.addEventListener('dragend', () => {
                tile.classList.remove('opacity-50', 'scale-95');
            });

            // 點擊事件 (適合點擊填空)
            tile.onclick = () => {
                if (selectedBopoElement) {
                    selectedBopoElement.classList.remove('selected-tile');
                }
                if (selectedBopoElement === tile) {
                    selectedBopoText = null;
                    selectedBopoElement = null;
                } else {
                    selectedBopoText = text;
                    selectedBopoElement = tile;
                    tile.classList.add('selected-tile');
                    playSound('select');
                    document.getElementById('feedback-message').innerHTML = `<span class="text-amber-600 font-bold animate-pulse"><i class="fa-solid fa-hand-pointer mr-1"></i> 已選取「${text}」，請點擊中間填空槽放入！</span>`;
                }
            };

            return tile;
        }

            }
        }

        /* 輔助：提取聲調標示與移除聲調 */
        function getToneMark(pinyinStr) {
            if (pinyinStr.includes('ˊ')) return 'ˊ';
            if (pinyinStr.includes('ˇ')) return 'ˇ';
            if (pinyinStr.includes('ˋ')) return 'ˋ';
            if (pinyinStr.includes('˙')) return '˙';
            return '';
        }

        function removeTone(pinyinStr) {
            return pinyinStr.replace(/[ˊˇˋ˙]/g, '');
        }

        /* 語音朗讀 (Web Speech API) */
        function speakCurrentWord() {
            if (!('speechSynthesis' in window)) return;
            window.speechSynthesis.cancel();
            const q = activeQuestions[currentIndex];
            const utterance = new SpeechSynthesisUtterance(q.word);
            utterance.lang = 'zh-TW';
            utterance.rate = 0.8;
            window.speechSynthesis.speak(utterance);
        }
